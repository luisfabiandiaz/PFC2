\begin{frame}{Contexto: ST-CC y el co-agrupamiento en tablero}
  \begin{columns}[c]
    \column{0.58\textwidth}
    \small
    \begin{itemize}
      \setlength{\itemsep}{1.2ex}
      \item Viajes en taxi, sismos y monitoreo ambiental describen \alert{dónde y cuándo} ocurre un
            fenómeno, y espacio y tiempo no son independientes \cite{Liu2021}.
      \item El \alert{co-agrupamiento} agrupa filas y columnas a la vez. \cite{Liu2021} proponen
            \STCC\ sobre una matriz de regiones $\times$ intervalos de tiempo, con \BBAC\
            \cite{Banerjee2007}.
      \item El resultado es un tablero de $K_R\times K_T$ bloques rectangulares. En Manhattan
            (noviembre de 2017) usaron unos 10 millones de viajes, 66 regiones y 12 intervalos de 2 horas.
      \item Es fácil de interpretar, pero la estructura es \alert{global}.
    \end{itemize}
    \column{0.38\textwidth}
    \centering
    \matriz{1}{0.34cm}\\[0.8ex]
    {\scriptsize Tablero de \STCC: cada bloque se resume con un valor}
  \end{columns}
\end{frame}

\begin{frame}{La limitación: un detalle local obliga a refinar todo}
  \centering
  \matriz{0}{0.31cm}\hspace{1.6cm}\matriz{1}{0.31cm}\\[0.5ex]
  {\scriptsize\hspace{0.2cm}(a) Datos: un foco local en un bloque \hspace{3.2cm} (b) Tablero de \STCC: el foco queda promediado}
  \vspace{1.2ex}
  \begin{itemize}
    \setlength{\itemsep}{0.6ex}
    \item Cada corte de filas o columnas atraviesa \alert{toda} la matriz, así que capturar el foco exige
          aumentar la resolución global y fragmentar zonas sin patrón.
    \item El paso con K-means de \cite{Liu2021} reagrupa bloques ya formados, pero no subdivide un bloque por dentro.
    \item Elegir $K_R$ y $K_T$ con una curva o con Cross-Associations \cite{Chakrabarti2004} sigue dando una rejilla global.
  \end{itemize}
\end{frame}

\begin{frame}{Planteamiento del problema}
  \begin{block}{Problema}
    ¿Cómo refinar \alert{localmente} y de forma \alert{estadísticamente justificada} la partición de
    \STCC\ para detectar patrones espacio-temporales locales, manteniendo la interpretabilidad del resultado?
  \end{block}
  \vspace{1ex}
  \begin{columns}[T]
    \column{0.31\textwidth}
    \begin{alertblock}{¿Dónde refinar?}
      \small No hay un criterio que diga si un bloque contiene heterogeneidad real que
      justifique subdividirlo.
    \end{alertblock}
    \column{0.31\textwidth}
    \begin{alertblock}{¿Cuándo detenerse?}
      \small Evaluar muchas divisiones sobre los mismos datos produce refinamientos
      espurios si no se controla la multiplicidad.
    \end{alertblock}
    \column{0.31\textwidth}
    \begin{alertblock}{¿Qué es ruido?}
      \small La matriz suma varios días con variabilidad mayor que la de Poisson, y con millones
      de viajes casi todo resulta significativo.
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{Objetivos}
  \begin{block}{Objetivo general}
    \small Proponer y validar una extensión de \STCC\ basada en un \alert{refinamiento jerárquico por
    bloques}, guiado por pruebas de significancia derivadas de la I-divergencia, que detecte patrones
    espacio-temporales locales y conserve la interpretabilidad del resultado.
  \end{block}
  \vspace{0.6ex}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{0.9ex}
    \item[\textbf{OE1}] Formular el refinamiento como un árbol de bloques anidados desde la partición
          base de \BBAC, con un criterio de heterogeneidad coherente con la función objetivo de \STCC.
    \item[\textbf{OE2}] Diseñar la regla de decisión y de parada: prueba global por bloque, mejor corte,
          calibración con la variabilidad entre días, control de multiplicidad y umbral de tamaño de efecto.
    \item[\textbf{OE3}] Implementar el método y evaluar su calibración y potencia con datos sintéticos y con
          datos nulos que reproducen la sobredispersión real.
    \item[\textbf{OE4}] Aplicarlo a los taxis de Manhattan y compararlo con \STCC\ y rejillas globales en
          error de reconstrucción por parámetro, estabilidad e interpretabilidad.
  \end{enumerate}
\end{frame}

\begin{frame}{Estado del arte: tres familias de trabajos desde 2020}
  \small
  \begin{columns}[T]
    \column{0.315\textwidth}
    \begin{block}{Co-agrupamiento y bloques flexibles}
      \footnotesize
      Co-clusters coherentes, biclustering con prueba de significancia, modelos de bloques bayesianos
      y jerárquicos \cite{Wu2022,Neves2022,Yen2020,Sugahara2023,Watanabe2021gof}.\\[0.5ex]
      \alert{Diferencia.} Tableros globales o pruebas como filtro posterior; ninguno decide por
      bloque cuándo subdividir.
    \end{block}
    \column{0.315\textwidth}
    \begin{block}{Inferencia selectiva}
      \footnotesize
      Pruebas válidas tras agrupar, \emph{count splitting} y \emph{data thinning}
      \cite{Gao2024,Chen2023,Watanabe2021si,Neufeld2024cs,Neufeld2024dt}.\\[0.5ex]
      \alert{Diferencia.} Validan estructuras ya estimadas; ninguno construye un refinamiento
      jerárquico local.
    \end{block}
    \column{0.315\textwidth}
    \begin{block}{Hotspots espacio-temporales}
      \footnotesize
      \emph{Space-time NetScan} y detección de focos urbanos en paradas de taxi
      \cite{Shiode2022,Chen2021hotspots}.\\[0.5ex]
      \alert{Diferencia.} Buscan zonas de alta intensidad en el espacio, no interacciones dentro de
      bloques de una matriz región~$\times$~tiempo.
    \end{block}
  \end{columns}
  \vspace{0.8ex}
  {\footnotesize\alert{Puntos de partida.} \STCC\ \cite{Liu2021} aporta la partición base. SigTree \cite{Hu2025} aporta
  el principio de construir la partición con pruebas de hipótesis y detenerla cuando ninguna división es significativa.\par}
\end{frame}

\begin{frame}{Calibración, multiplicidad y parada}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \small
    \begin{itemize}
      \setlength{\itemsep}{0.9ex}
      \item \alert{Bootstrap de días.} La varianza entre días es 3 veces la media en días laborables y
            17 veces en feriados. Se remuestrean los días, lo que conserva la correlación entre celdas.
      \item \alert{Ajuste min-P} \cite{Westfall1993}. Se repite la búsqueda del mejor corte en cada réplica nula.
            Sin ello, el error de tipo~I llegó a $0.98$.
      \item \alert{Nivel por masa.} $\alpha_C=\alpha_B N_C/N_B$ a lo largo del árbol acota el error por
            familia por $\alpha$.
      \item \alert{Parada} por significancia \alert{y} tamaño de efecto $\tilde I_B\ge\delta$, que no crece
            con el número de viajes.
    \end{itemize}
    \column{0.46\textwidth}
    \centering\scriptsize
    Error de tipo~I al nivel $0.05$ (datos nulos semirreales, media por bloque)\\[2ex]
    \begin{tikzpicture}[x=4.1cm,y=0.5cm]
      \barra{0}{Poisson}{0.40}{0.76}

    \tiny Con pocos días (feriado), el bootstrap es algo liberal en bloques nocturnos. Es una limitación conocida.
  \end{columns}
\end{frame}
